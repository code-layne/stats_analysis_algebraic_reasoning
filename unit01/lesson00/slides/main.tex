% ── DEBRIEF 1 ────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Debrief: naming what you found}
  \sectionlabel[redacc]{1. a number, and a table}

  \vspace{0.3cm}
  \begin{center}
    {\Huge\bfseries 12} \qquad vs. \qquad
    {\large ``12 points, for the six players on the Riverbend roster''}
  \end{center}

  \vspace{0.4cm}
  \begin{center}
  \begin{minipage}{0.86\textwidth}
    \begin{itemize}\setlength{\itemsep}{7pt}
      \item A number plus \textbf{who}, \textbf{what}, and \textbf{units} is
            \redink{data}. Without them it is just a number.
      \item Each \textbf{row} of the roster is one \redink{individual} --- 6 of them.
      \item Each \textbf{column} is one \redink{variable} --- 5 of them.
            (``Player'' names the individuals; it is not a variable we study.)
    \end{itemize}
  \end{minipage}
  \end{center}
\end{frame}

% ── DEBRIEF 3 (the crux) ─────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Debrief: the blog's arithmetic was right}
  \sectionlabel[redacc]{3. the test to write down}

  \vspace{0.2cm}
  \begin{center}
    $\dfrac{12+7+23+4+17+21}{6} \;=\; \dfrac{84}{6} \;=\; 14$ \qquad
    {\color{goldacc}\bfseries Correct.}
  \end{center}

  \vspace{0.3cm}
  \begin{center}
    {\color{goldacc}\bfseries\large So who wears 14 on average?}
  \end{center}

  \vspace{0.3cm}
  \begin{center}
    Nobody. Jersey \# was in your \textbf{label} pile --- the digits only tell the players
    apart, so the mean measures nothing at all.
  \end{center}

  \vspace{0.35cm}
  \begin{block}{The test --- write this down}
    Not ``does it look like a number?'' but
    \redink{does arithmetic on this variable mean anything?}\\[3pt]
    Grade level and ZIP code fail it too. \textbf{Siblings} $2,0,3,1,2,4$ passes it ---
    digits are not the enemy, meaningless arithmetic is.
  \end{block}
\end{frame}

% ── DEBRIEF 4 ────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Debrief: which questions take data?}
  \sectionlabel[redacc]{4. one answer, or many}

  \vspace{0.2cm}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{One look-up (O)}
        ``What number does Sasha wear?''\\
        ``How tall is Marcus?''\\[4pt]
        Look at one row. One answer. Done.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Many answers (M)}
        ``How many points do Riverbend players score in a game?''\\
        ``How tall are the players on the roster?''\\[4pt]
        The answers \textbf{vary} from player to player.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.5cm}
  \begin{center}
    Your \textbf{M} questions have a name: a \redink{statistical question} is one whose
    answers are \textbf{expected to vary}, so it takes data to answer.\\[6pt]
    {\color{cerulean}\bfseries A question can contain a number and still not be statistical.}
  \end{center}
\end{frame}

% ── QUICKNOTES ───────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{QuickNotes: naming what you found}
  \sectionlabel{Fill the box in your packet}
  \vspace{0.2cm}
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item \textbf{Data} = numbers or labels $+$ the \redink{context}: who, what, and the
          \redink{units}.
    \item Row $=$ one \redink{individual}. \qquad Column $=$ one \redink{variable}.
    \item \redink{Quantitative} $\rightarrow$ measured amount with units
          $\rightarrow$ summarize with a \redink{mean}.
    \item \redink{Categorical} $\rightarrow$ label or group name $\rightarrow$ summarize with
          counts and \redink{percents}.
    \item The test for digits: \redink{does arithmetic on it mean anything?}
    \item A \redink{statistical question} expects the answers to \redink{vary}.
  \end{itemize}
\end{frame}

  \sectionlabel[goldacc]{We work this one together --- you hold the pen}

  \begin{center}
    \footnotesize
    \renewcommand{\arraystretch}{1.1}
    \begin{tabular}{@{} l c c l c c @{}}
    \hline
    \textbf{Student} & \textbf{Grade} & \textbf{Commute (min)} & \textbf{Lunch} &
    \textbf{Siblings} & \textbf{Club} \\
    \hline
    Alina   & 10 & 20 & Home       & 2 & Yes \\
    Devon   & 11 &  8 & Cafeteria  & 0 & No  \\
    Marisol &  9 & 35 & Home       & 3 & Yes \\
    Theo    & 12 & 15 & Cafeteria  & 1 & Yes \\
    Priya   & 10 &  5 & Off campus & 2 & No  \\
    Jamal   & 11 & 25 & Cafeteria  & 4 & Yes \\
    \hline
    \end{tabular}
  \end{center}

  \vspace{0.05cm}
  \begin{center}
  \begin{minipage}{0.9\textwidth}
    \begin{itemize}\setlength{\itemsep}{4pt}
      \item How many individuals? How many variables? Classify Commute, Lunch, and
            \textbf{Grade}.
      \item Mean commute: $\dfrac{20+8+35+15+5+25}{6} = \dfrac{108}{6} = 18$ minutes.
            \textbf{Interpret it} --- name the group and the units.
      \item \textbf{What if} we recorded \emph{minutes waiting in the lunch line} instead
            of \emph{where} they eat? Does the type change? \textbf{Justify.}
    \end{itemize}
  \end{minipage}
  \end{center}
\end{frame}
