% ── NOTES 1: DATA NEED CONTEXT ───────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Data are numbers \emph{with a context}}
  \sectionlabel{notes section 1}

  \vspace{0.5cm}
  \begin{center}
    {\Huge\bfseries 12}
  \end{center}

  \vspace{0.4cm}
  \begin{center}
    What does that tell you? \quad {\color{cerulean}\bfseries Nothing.}
  \end{center}

  \vspace{0.5cm}
  \begin{block}{Now add the context --- who, what, units}
    ``The \textbf{six players on the Riverbend varsity roster} scored a mean of
    \textbf{12 points} per player.''
  \end{block}

  \vspace{0.3cm}
  \begin{center}
    {\color{goldacc}\bfseries That sentence is data. The bare 12 is not.}
  \end{center}
\end{frame}

% ── NOTES 3: TWO TYPES ───────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Two types of variables}
  \sectionlabel{notes section 3}

  \vspace{0.4cm}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{Quantitative}
        Records a \textbf{measured amount}, with units.\\[5pt]
        Points, Height, minutes, siblings.\\[5pt]
        {\color{cerulean}\bfseries Summarize with a mean.}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Categorical}
        Records a \textbf{label} or group name.\\[5pt]
        Position, Captain, lunch choice.\\[5pt]
        {\color{cerulean}\bfseries Summarize with counts and percents.}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.7cm}
  \begin{center}
    Captains: $3$ of $6 = \mathbf{50\%}$. \quad
    {\color{goldacc}\bfseries A categorical variable does get summarized --- just not
    with a mean.}
  \end{center}
\end{frame}

% ── NOTES 4: THE DIGIT TEST (the crux) ───────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{The blog's arithmetic was right}
  \sectionlabel[redacc]{notes section 4 --- the digit test}

  \vspace{0.3cm}
  \begin{center}
    $\dfrac{12+7+23+4+17+21}{6} = \dfrac{84}{6} = 14$
  \end{center}

  \vspace{0.5cm}
  \begin{center}
    {\color{redacc}\bfseries\large Now find me the player who wears 14 on average.}
  \end{center}

  \vspace{0.6cm}
  \begin{block}{The test is not ``does it look like a number?''}
    \begin{center}
      \textbf{Does \emph{arithmetic} on this variable mean anything?}
    \end{center}
  \end{block}

  \vspace{0.3cm}
  \begin{center}
    Fails it: jersey number, ZIP code, grade level. \quad
    Passes it: points, height, minutes.
  \end{center}
\end{frame}

% ── NOTES 5: STATISTICAL QUESTIONS ───────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Which questions take data to answer?}
  \sectionlabel{notes section 5}

  \vspace{0.3cm}
  \begin{center}
    A \textbf{statistical question} is one whose answers are expected to
    {\color{cerulean}\bfseries vary} from individual to individual.
  \end{center}

  \vspace{0.5cm}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{One look-up (O)}
        What number does Sasha wear?\\[4pt]
        How tall is Marcus?
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Takes many answers (M)}
        How many points do Riverbend players score in a game?\\[4pt]
        How tall are the players on the roster?
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.5cm}
  \begin{center}
    {\color{goldacc}\bfseries The M questions are the statistical ones.}
  \end{center}
\end{frame}

% ── DEBRIEF ──────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Debrief --- correct your reasons in a second color}
  \sectionlabel[redacc]{7 minutes}

  \vspace{0.3cm}
  \begin{center}
    \begin{minipage}{0.88\textwidth}
      \begin{enumerate}
        \setlength{\itemsep}{7pt}
        \item A number is not an answer until it names a \textbf{group} and
              \textbf{units}.
        \item The recap blog's mean bib number of $23$ is arithmetically correct
              --- and worthless.
        \item State the digit test in your own words.
        \item Same runners, two questions, two different types. \textbf{The
              question decides the type.}
      \end{enumerate}
    \end{minipage}
  \end{center}

  \vspace{0.5cm}
  \begin{center}
    {\color{redacc}\bfseries A correct computation on the wrong kind of variable is
    still a worthless number.}
  \end{center}
