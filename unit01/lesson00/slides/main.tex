% ── TWO TYPES ────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Two types of variables}
  \sectionlabel{one question decides it}

  \begin{center}
    {\large\bfseries Could I find an average of this column?}
  \end{center}

  \vspace{0.4cm}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{Commute (min)}
        $20,\; 8,\; 35,\; 15,\; 5,\; 25$\\[4pt]
        Mean $= 18$ minutes. That is a real commute length.\\[4pt]
        {\color{cerulean}\bfseries Quantitative} --- a measured amount with units.
        Summarize with a \textbf{mean}.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Lunch}
        Home, Cafeteria, Home, Cafeteria, Off campus, Cafeteria\\[4pt]
        There is no average of three place names.\\[4pt]
        {\color{cerulean}\bfseries Categorical} --- a label.
        Summarize with \textbf{counts and percents}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ── THE TRAP ─────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{The trap: numbers that are really labels}
  \sectionlabel[goldacc]{grade level}

  \begin{center}
    $10, \quad 11, \quad 9, \quad 12, \quad 10, \quad 11$
  \end{center}

  \vspace{0.3cm}
  \begin{center}
    It is made of digits. Does that make it quantitative?
  \end{center}

  \vspace{0.3cm}
  \begin{center}
    $\dfrac{10+11+9+12+10+11}{6} \; = \; \dfrac{63}{6} \; = \; 10.5$
  \end{center}

  \vspace{0.3cm}
  \begin{center}
    {\color{goldacc}\bfseries\large Who is in grade 10.5?}
  \end{center}

  \vspace{0.4cm}
  \begin{block}{The test --- write this down}
    Not ``does it look like a number?'' but
    \textbf{does arithmetic on this variable mean anything?}\\[3pt]
    Jersey numbers and ZIP codes fail the same test.
  \end{block}
\end{frame}

% ── STATISTICAL QUESTIONS ────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Statistical questions}
  \sectionlabel{variability is the point}

  \vspace{0.2cm}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{Not statistical}
        ``How tall am I?''\\[4pt]
        Measure once. One answer. Done.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Statistical}
        ``How tall are students at Riverbend?''\\[4pt]
        The answers \textbf{vary} from student to student, so it takes data.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.5cm}
  \begin{center}
    \begin{minipage}{0.85\textwidth}
      \textbf{S or N?}
      \begin{itemize}
        \setlength{\itemsep}{4pt}
        \item How many students are enrolled at Riverbend?
        \item How long is the commute for students at Riverbend?
        \item What did Devon eat for lunch today?
        \item Where do Riverbend students eat lunch?
      \end{itemize}
    \end{minipage}
  \end{center}
\end{frame}

% ── CLOSE ────────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Today in one line}
  \sectionlabel[goldacc]{exit ticket next}

  \vspace{0.6cm}
  \begin{center}
    \begin{minipage}{0.85\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item Data are numbers \textbf{with a context}: who, what, units.
        \item Rows are \textbf{individuals}; columns are \textbf{variables}.
        \item \textbf{Quantitative} $\rightarrow$ mean.
              \textbf{Categorical} $\rightarrow$ counts and percents.
        \item Digits do not make a variable quantitative ---
              \textbf{averaging has to mean something}.
        \item A \textbf{statistical question} expects the answers to vary.
      \end{itemize}
    \end{minipage}
  \end{center}
