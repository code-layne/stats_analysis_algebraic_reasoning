\begin{notesbox}{Exit Ticket --- No Notes}
\small
\begin{enumerate}[leftmargin=1.6em, itemsep=10pt, topsep=3pt]

  \item A music teacher records two things about each student in the school
        orchestra: the \textbf{instrument} they play and the \textbf{number of
        years} they have played it.

        Classify each variable and justify with the averaging test.

        \begin{center}
        \renewcommand{\arraystretch}{1.5}
        \begin{tabularx}{0.95\linewidth}{@{} p{3cm} p{3cm} X @{}}
        \toprule
        \textbf{Variable} & \textbf{Type} & \textbf{How I know} \\
        \midrule
        Instrument      & \ans{categorical}  & \ans{a name, not an amount --- no mean} \\
        Years played    & \ans{quantitative} & \ans{a count of years; a mean makes sense} \\
        \bottomrule
        \end{tabularx}
        \end{center}

  \item Only one of these is a statistical question. Circle it, then explain your
        choice.

        \begin{itemize}[leftmargin=1.6em, itemsep=3pt, topsep=3pt]
          \item[\textbf{A.}] How many instruments does the school own?
          \item[\textbf{B.}] How many years have orchestra students played their
                instrument? \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
        \end{itemize}

        Why? \ansline{B's answers vary from student to student; A has one fixed answer, so no data are needed.}

  \item A reporter writes: \textit{``The mean of the numbers on the team's
        jerseys is $42$.''}

        What does this tell you about the team? Explain.
        \ansline{Nothing. Jersey number is a label written with digits, not a measured amount,}
        \ansline{so averaging it produces a number that describes no real quantity.}

\end{enumerate}
\end{notesbox}
