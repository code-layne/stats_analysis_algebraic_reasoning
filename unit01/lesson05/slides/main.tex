% ── WHAT IS AN OBSERVATIONAL STUDY ───────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Watching, not assigning}
  \sectionlabel{what makes it observational}

  \vspace{0.3cm}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{Observational study}
        You measure or survey a sample \textbf{without trying to affect
        anyone}. The groups already exist --- the members chose their own swim
        time. A survey is one type.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Not observational}
        ``Assign half the members to 6~a.m.\ and half to 7~p.m.\ for a month.''
        Somebody \textbf{put people into groups}. That is an experiment ---
        next class.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.7cm}
  \begin{center}
    \begin{minipage}{0.85\textwidth}
      \small The whole difference is one word: \textbf{assign}. Every study in
      Unit 1 so far has been observational.
    \end{minipage}
  \end{center}
\end{frame}

    \renewcommand{\arraystretch}{1.35}
    \begin{tabular}{@{} l l @{}}
    \hline
    \textbf{Formulate} & Do morning swimmers sleep more than evening

    \textbf{Collect} & Random sample of 200 from the full 1,500-member list \\
    \textbf{Represent} & Two-way table of swim time vs.\ sleep, one percent

    \hline
    \end{tabular}
  \end{center}

  \begin{center}
    {\color{cerulean}\bfseries Nowhere in the plan do we tell a single member
    when to swim.}
  \end{center}
\end{frame}

% ── THREE STORIES ────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Three stories fit this table}
  \sectionlabel[goldacc]{association is not causation}

  \vspace{0.4cm}
  \begin{center}
    \begin{minipage}{0.88\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item \textbf{The claim:} morning swim $\rightarrow$ more sleep.
        \item \textbf{The arrow backwards:} people who already sleep well are
              the ones who can get up at 6~a.m.
        \item \textbf{Something else moves both:} 75\% of the morning swimmers
              are retired --- against 15\% of the evening swimmers.
      \end{itemize}
    \end{minipage}
  \end{center}

  \vspace{0.6cm}
  \begin{center}
    {\color{goldacc}\bfseries A variable outside the study that moves both
    variables in it is a lurking variable.}
  \end{center}
\end{frame}

% ── WHAT YOU MAY CLAIM ───────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{What you may say, and what you may not}
  \sectionlabel{writing it up}

  \vspace{0.3cm}
  \begin{center}
    \small
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{@{} l c @{}}
    \hline
    \textbf{Statement} & \textbf{Supported?} \\
    \hline
    Morning swimmers report more sleep than evening swimmers & \textbf{yes} \\
    Swimming in the morning makes a person sleep longer & \textbf{no} \\
    If evening swimmers switched, they would sleep more & \textbf{no} \\
    \hline
    \end{tabular}
  \end{center}

  \vspace{0.6cm}
  \begin{center}
    \begin{minipage}{0.85\textwidth}
      \small Report both percents and the gap, and say the variables are
      \textbf{associated}. Never write \emph{causes}, \emph{makes},
      \emph{raises}, or \emph{improves}.
    \end{minipage}
  \end{center}
\end{frame}

% ── CLOSE ────────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Today in one line}
  \sectionlabel[goldacc]{exit ticket next}

  \vspace{0.5cm}
  \begin{center}
    \begin{minipage}{0.86\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item An \textbf{observational study} watches. Nobody is
              \textbf{assigned} to a group.
        \item You plan it with the same \textbf{statistical cycle} --- what
              changes is what you may conclude.
        \item Two variables moving together are \textbf{associated}. That is
              all you may report.
        \item An association has three explanations: the claim, the arrow
              backwards, or a \textbf{lurking variable}.
        \item \textbf{No bias does not mean causation.} A clean study still
              cannot tell you why.
      \end{itemize}
    \end{minipage}
  \end{center}
\end{frame}
