% ── CONSTRAINTS ──────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Constraints --- why we sample at all}
  \sectionlabel{the census is off the table}

  \vspace{0.3cm}
  \begin{center}
    \begin{minipage}{0.86\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{6pt}
        \item \textbf{Time} --- reaching everyone takes longer than you have.
        \item \textbf{Cost} --- reaching everyone costs more than the budget.
        \item \textbf{Access} --- some individuals cannot be found or reached.
        \item \textbf{Destruction} --- measuring ruins the item you measured.
        \item \textbf{Change} --- the population shifts while you collect.
      \end{itemize}
    \end{minipage}
  \end{center}

  \vspace{0.5cm}
  \begin{center}
    \small A cereal plant filled 12,000 boxes Tuesday. To weigh a box you have to
    open it.
  \end{center}
\end{frame}

% ── SAMPLING VARIABILITY ─────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Three volunteers. One Saturday. Three answers.}
  \sectionlabel[goldacc]{which one is wrong?}

  \vspace{0.2cm}
  \begin{center}
    \small
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{@{} l c l @{}}
    \hline
    \textbf{Volunteer} & \textbf{Drove, of 50} & \textbf{Percent} \\

    Ana  & 26 & $26/50 = 0.52 = 52\%$ \\
    Ben  & 24 & $24/50 = 0.48 = 48\%$ \\

    \hline

  \end{center}

  \vspace{0.5cm}
  \begin{center}
    Three statistics. \textbf{One} parameter. Nobody miscounted.\\[6pt]
    {\color{cerulean}\bfseries This is sampling variability --- it is what samples
    do.}
  \end{center}
\end{frame}

% ── WHAT YOU MAY CLAIM ───────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{What the sample lets you say}
  \sectionlabel{and what it does not}

  \vspace{0.3cm}
  \begin{block}{Allowed}
    ``About half of the shoppers \textbf{we asked} drove.''
  \end{block}

  \vspace{0.3cm}
  \begin{block}{Not allowed}
    ``\textbf{Exactly} 416 Saturday shoppers drove.'' --- 416 is an estimate, not
    a count.\\[4pt]
    ``52\% of \textbf{Lakeside residents} drive to the market.'' --- wrong
    population.
  \end{block}

  \vspace{0.4cm}
  \begin{center}
    {\color{goldacc}\bfseries Change the population and the same statistic
    estimates a different parameter.}
  \end{center}
\end{frame}

% ── CLOSE ────────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Today in one line}
  \sectionlabel[goldacc]{exit ticket next}

  \vspace{0.6cm}
  \begin{center}
    \begin{minipage}{0.85\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item The \textbf{population} is everyone the question is about; the
              \textbf{sample} is who you measured.
        \item A \textbf{parameter} describes the population and is almost always
              unknown.
        \item A \textbf{statistic} describes the sample --- it is the number you
              can actually compute.
        \item \textbf{Constraints} --- time, cost, access, destruction, change ---
              are why we sample.
        \item Two honest samples give two statistics for \textbf{one} parameter.
              That is not an error.
      \end{itemize}
    \end{minipage}
  \end{center}
\end{frame}
