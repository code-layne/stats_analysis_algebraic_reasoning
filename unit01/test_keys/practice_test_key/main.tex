\parthead{Part C --- Short Computations \hfill 4 questions, 10 points each}

\small
\begin{enumerate}[leftmargin=2.2em, itemsep=10pt, topsep=2pt, start=16]

% ---------------------------------------------------------------- C1 (1.2)
\item \textbf{Silver Lake Recreation Center} has $900$ members. The director asks
$60$ members chosen at random, and $18$ of them use the pool every week.

\vspace{2pt}
(a) What percent of the sample uses the pool every week?

\begin{work}
  \dfrac{18}{60} &= 0.30 = 30\%
\end{work}

(b) About how many of all $900$ members use the pool every week?

\begin{work}
  0.30 \times 900 &= 270 \text{ members}
\end{work}

% ---------------------------------------------------------------- C2 (1.3)
\boxguard[14]
\item \textbf{Granite Falls Middle School} has $600$ students and wants a
\textbf{stratified} sample of $60$ --- the same fraction from each grade.

\vspace{2pt}
(a) What fraction of the school is being sampled?

\begin{work}
  \dfrac{60}{600} &= \dfrac{1}{10}
\end{work}

(b) Complete the table.

\vspace{2pt}
\renewcommand{\arraystretch}{1.2}
\begin{tabularx}{\linewidth}{@{} X c c c | c @{}}
\rowcolor{frost}
\textbf{Grade} & 6 & 7 & 8 & \textbf{Total} \\
\midrule
Students in grade & $240$ & $200$ & $160$ & $600$ \\
Number sampled & \ans{$24$} & \ans{$20$} & \ans{$16$} & \ans{$60$} \\
\end{tabularx}

% ---------------------------------------------------------------- C3 (1.3)
\boxguard[14]
\item \textbf{Brookside Apartments} has $500$ households on a numbered list. The
manager wants a \textbf{systematic} sample of $50$.

\vspace{2pt}
(a) Find the interval $k$.

\begin{work}
  k &= 500 \div 50 = 10
\end{work}

(b) The random start is household $3$. List the first four households chosen.

\vspace{2pt}
\ans{$3$} \quad \ans{$13$} \quad \ans{$23$} \quad \ans{$33$}

% ---------------------------------------------------------------- C4 (1.6)
\boxguard[14]
\item In an \textbf{experiment}, $40$ students are randomly assigned to use a
flashcard app and $40$ to study the usual way. Afterwards, $30$ of the app group and
$20$ of the usual group passed a quiz.

\vspace{2pt}
(a) Find the percent who passed in each group.

\begin{work}
  \text{App: } \dfrac{30}{40} &= 0.75 = 75\% \\
  \text{Usual: } \dfrac{20}{40} &= 0.50 = 50\%
\end{work}

(b) How many percentage points apart are the two groups? \ans{$25$ points}

\end{enumerate}

\parthead{Part D --- Critical Thinking \hfill 1 question, 20 points}

\small
\begin{enumerate}[leftmargin=2.2em, itemsep=10pt, topsep=2pt, start=20]

\item A random sample of Maplewood students found that students who \textbf{eat
breakfast} have higher grades than students who skip it. Nobody was assigned ---
students chose for themselves. The principal says, \emph{``Eating breakfast causes
higher grades.''}

\vspace{3pt}
Is the principal's conclusion justified? Explain your reasoning, and name one
\textbf{other variable} that could explain the result.

\par\ansline{No. Nobody was assigned --- students chose --- so this is an observational study.}
\par\ansline{It shows an association between breakfast and grades, not causation.}
\par\ansline{Another variable: family routine. Students with more support at home are more}
\par\ansline{likely to eat breakfast \emph{and} to keep up with homework,}
\par\ansline{and that support could raise grades by itself.}
\par\ansline{Only an experiment that randomly assigns breakfast could show it causes higher grades.}

\end{enumerate}
