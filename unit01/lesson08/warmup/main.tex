\begin{notesbox}{Spiral Review --- A Study That Is Already Finished}
\small
The Riverbend Student Council has a \textbf{\$3{,}000} activities grant to spend
and asked the school which one thing it should fund. It drew a
\textbf{stratified sample of $60$} from the $900$ students and got one answer
from every student it drew.

\vspace{3pt}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{0.9\linewidth}{@{} X c c c c @{}}
\toprule
\textbf{Choice} & Club funding & Field trips & Gym equipment & Longer lunch \\
\midrule
\textbf{Students} & $21$ & $18$ & $15$ & $6$ \\
\bottomrule
\end{tabularx}
\end{center}

\begin{enumerate}[leftmargin=1.5em, itemsep=8pt, topsep=5pt]

  \item What \textbf{percent} of the $60$ students chose club funding?

        \begin{work}
          \dfrac{21}{60} &= 0.35 \\
                         &= 35\%
        \end{work}

  \item What \textbf{percent} chose a longer lunch?

        \begin{work}
          \dfrac{6}{60} &= 0.10 \\
                        &= 10\%
        \end{work}

  \item Use the sample to speak about the whole school.

        \textbf{(a)} About how many of all $900$ students would choose club
        funding?

        \begin{work}
          0.35 \times 900 &= 315 \text{ students}
        \end{work}

        \textbf{(b)} The four percents add to $100\%$. What does that tell you
        about the survey question the council asked?
        \writeline

\end{enumerate}
\end{notesbox}
