% ── THE QUESTION ─────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Stage 1: does your question pass?}
  \sectionlabel{three tests}

  \vspace{0.5cm}
  \begin{center}
    \begin{minipage}{0.88\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item The answers \textbf{vary} --- otherwise you already know.
        \item A \textbf{group} is named --- whom is this about?
        \item A \textbf{measurement} is named --- what exactly gets recorded?
      \end{itemize}
    \end{minipage}
  \end{center}

  \vspace{0.5cm}
  \begin{center}
    {\color{goldacc}\bfseries ``How should we spend the grant?'' fails two of
    the three. Repair it before you ask anyone.}
  \end{center}
\end{frame}

% ── THE INSTRUMENT ───────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Stage 2: writing the instrument}
  \sectionlabel[goldacc]{AFDA.DA.2d}

  \vspace{0.4cm}
  \begin{center}
    \begin{minipage}{0.88\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{7pt}
        \item \textbf{One idea per item.} ``Clubs \emph{and} field trips?''
              cannot be answered.
        \item \textbf{Neutral wording.} ``Don't you agree\dots'' tells them the
              answer.
        \item \textbf{Choices cover everyone, no overlap.} Then percents add to
              100\%.
        \item \textbf{Short.} People answer what they finish reading.
      \end{itemize}
    \end{minipage}
  \end{center}

  \vspace{0.4cm}
  \begin{center}
    {\color{cerulean}\bfseries A biased instrument is wrong in a direction ---
    and arithmetic cannot pull it back.}
  \end{center}
\end{frame}

% ── CLOSE ────────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Unit 1 in one line}
  \sectionlabel[goldacc]{exit ticket next}

  \vspace{0.5cm}
  \begin{center}
    \begin{minipage}{0.86\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{8pt}
        \item A survey is an \textbf{observational study} --- plan it with the
              cycle.
        \item The \textbf{instrument} decides the data; bias written into it
              stays in.
        \item Categorical data $\Rightarrow$ frequency table $\Rightarrow$
              \textbf{bar graph}.
        \item A percent scales to the population only if \textbf{chance} chose
              the sample.
        \item A conclusion is only as trustworthy as the study that produced
              it.
      \end{itemize}
    \end{minipage}
  \end{center}
