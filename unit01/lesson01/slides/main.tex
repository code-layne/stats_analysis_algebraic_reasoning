% ── THE FOUR STAGES ──────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{The data cycle --- four stages, in order}
  \sectionlabel{and then it starts over}

  \vspace{0.2cm}
  \begin{center}
    \small
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{@{} c l l @{}}
    \hline
    \textbf{Stage} & \textbf{Name} & \textbf{What it hands to the next stage} \\
    \hline
    1 & Formulate questions      & A question naming a group and a measurement \\
    2 & Collect or acquire data  & The raw values, one per individual \\
    3 & Organize and represent   & A table that makes the pattern visible \\
    4 & Analyze and communicate  & A conclusion in context --- and a new question \\
    \hline
    \end{tabular}
  \end{center}

  \vspace{0.5cm}
  \begin{center}
    {\color{cerulean}\bfseries Each stage can only use what the stage before it
    handed over.}
  \end{center}
\end{frame}

% ── STAGE 1: THE QUESTION ────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Stage 1 --- a question data can answer}
  \sectionlabel{three tests}

  \vspace{0.3cm}
  \begin{center}
    \begin{minipage}{0.8\textwidth}
      \begin{enumerate}
        \setlength{\itemsep}{6pt}
        \item Do the answers \textbf{vary} from one individual to the next?
        \item Does it name the \textbf{group} it is about?
        \item Does it say what will be \textbf{measured} or recorded?
      \end{enumerate}
    \end{minipage}
  \end{center}

  \vspace{0.5cm}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{Not ready}
        ``How do I get to school?''\\[3pt]
        ``Do people like biking?''
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Ready}
        ``How do \textbf{Riverbend students} usually \textbf{get to school}?''
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ── THE QUESTION DECIDES THE TYPE ────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{The question decides the type of data}
  \sectionlabel{before any data arrive}

  \vspace{0.3cm}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{``How do students get to school?''}
        Record: the \textbf{way} each student travels.\\[4pt]
        {\color{cerulean}\bfseries Categorical} ---
        summarize with counts and percents.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{``How many minutes is the trip?''}
        Record: the \textbf{minutes} each trip takes.\\[4pt]
        {\color{cerulean}\bfseries Quantitative} ---
        summarize with a mean.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.7cm}
  \begin{center}
    \begin{minipage}{0.85\textwidth}
      If you cannot finish ``for each student I will record
      \underline{\hspace{2.5cm}},'' the question is not finished either.
    \end{minipage}
  \end{center}
\end{frame}

% ── THE CYCLE TURNS ──────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{The cycle turns}
  \sectionlabel[goldacc]{which stage broke?}

  \vspace{0.2cm}
  \begin{center}
    The answer raised a new question --- so we go back to stage 1.
  \end{center}

  \vspace{0.5cm}
  \begin{center}
    \begin{minipage}{0.88\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{7pt}
        \item Only students at the bike rack were surveyed.
        \item The report says ``most students want racks,'' but nobody was asked
              about racks.
        \item The question never said \emph{which} students.
        \item The 60 answers were left as a list nobody could read.
      \end{itemize}
    \end{minipage}
  \end{center}
\end{frame}

% ── CLOSE ────────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Today in one line}
  \sectionlabel[goldacc]{exit ticket next}

  \vspace{0.6cm}
  \begin{center}
    \begin{minipage}{0.85\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item Four stages, in order --- and each one feeds the next.
        \item A question is ready when the answers \textbf{vary}, the
              \textbf{group} is named, and the \textbf{measurement} is named.
        \item The \textbf{question} decides whether the data are categorical or
              quantitative.
        \item Counts become percents; percents scale up to the population.
        \item A conclusion describes \textbf{only} the group you asked, and
              answers \textbf{only} the question you asked.
      \end{itemize}
    \end{minipage}
  \end{center}
\end{frame}
