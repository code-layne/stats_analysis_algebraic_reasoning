% ── WHEN YOU CANNOT ASSIGN ───────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{When you cannot run an experiment}
  \sectionlabel[goldacc]{three reasons}

  \vspace{0.4cm}
  \begin{center}
    \begin{minipage}{0.88\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item \textbf{Unethical} --- assigning it would harm someone. You
              cannot assign anyone to smoke.
        \item \textbf{Impossible} --- the treatment is something a person
              already \emph{is} or \emph{has}: retired, arthritic, a
              45-minute bus ride.
        \item \textbf{Impractical} --- too costly, too slow, or nobody would
              agree to be assigned.
      \end{itemize}
    \end{minipage}
  \end{center}

  \vspace{0.5cm}
  \begin{center}
    {\color{goldacc}\bfseries Then you run the best observational study you
    can --- and you weaken the verb.}
  \end{center}
\end{frame}

% ── THE TRAP ─────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Observation is not ``observational study''}
  \sectionlabel{the one to get wrong}

  \vspace{0.5cm}
  \begin{center}
    \begin{minipage}{0.88\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{Observation} is a \emph{method} --- how the data get
              recorded.
        \item An \textbf{observational study} is a \emph{design} --- nobody
              assigned the groups.
        \item Bayside's reading experiment collected its data by observation.
              It was still an experiment.
      \end{itemize}
    \end{minipage}
  \end{center}

  \vspace{0.6cm}
  \begin{center}
    {\color{goldacc}\bfseries Study type: who built the groups. Method: what
    the question needs.}
  \end{center}
\end{frame}

% ── CLOSE ────────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \ceruleanheader{Today in one line}
  \sectionlabel[goldacc]{exit ticket next}

  \vspace{0.5cm}
  \begin{center}
    \begin{minipage}{0.86\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item Observational: groups already exist $\Rightarrow$
              \textbf{association}.
        \item Experiment: chance assigns $\Rightarrow$ \textbf{cause and
              effect}.
        \item You can only assign when it is ethical, possible, and practical.
        \item The \textbf{statistical cycle} plans an experiment too --- the
              assigning lives in \emph{Collect}.
        \item Then choose the \textbf{method}: survey, interview, focus group,
              observation, content analysis.
      \end{itemize}
    \end{minipage}
  \end{center}
